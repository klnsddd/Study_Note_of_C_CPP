`fstream`：C++中最常用的一个文件操作类
	常用的一些成员函数
		`open()`    `close()`    `is_open()`    `eof()`    `put()`     `get()`
		`write()`    `read()`    `<<`    `>>`

	使用重载的`<<` `>>`


```C++
#include <iostream>
using namespace std;
#include <fstream>


int main()
{
	fstream obj;
	
	obj.open("test_1.txt", ios::out);
	obj.put('S');
	char ch = 'A';
	obj.put(ch);
	obj.close();
	
	obj.open("test_1.txt", ios::in);
	obj.get(ch);
	cout << ch << endl;
	ch = obj.get();
	cout << ch << endl;
	obj.close();
	
	//obj.is_open(); //是否成功打开文件
	//obj.eof(); //是否到达文件尾
	
	//二进制读写，效率较高
	obj.open("test_2.txt", ios::out);
	int num = 99;
	obj.write((const char*)&num, sizeof(int));
	obj.close();
	
	obj.open("test_2.txt", ios::in);
	int val = 0;
	obj.read((char*)&val, sizeof(int));
	obj.close();
	cout << "val = " << val << endl;
	
	//使用重载的 << >>
	obj.open("test_3.txt", ios::out);
	obj << "asdfasd!!" << endl;
	obj << "123" << endl;
	obj.close();
	
	char str[64];
	int num = 0;
	obj.open("test_3.txt", ios::in);
	obj >> str;
	obj >> num;
	obj.close();
	cout << str << endl;
	cout << num << endl;
	
	return 0;
}
```
