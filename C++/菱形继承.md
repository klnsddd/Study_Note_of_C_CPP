类AA继承两份类A的成员
容易造成内存浪费
- 虚继承 `virtual`，继承时会额外添加四个字节来存储虚指针，但菱形继承出的子类不会再出现成员重复

```C++
#include <iostream>
using namespace std;
class A
{
	int n;
};

class A_1 : virtual public A
{
	int a1;
};
class A_2 : virtual public A
{
	int a2;
};

class AA : public A_1, public A_2 //A_1,A_2虚继承后，AA就不会有重复的成员了
{
	int aa;
};


int main()
{
	AA obj_aa;
	obj_aa.aa;
	obj_aa.a1;
	obj_aa.a2;
	obj_aa.n;  //在虚继承的情况下就可以使用了
	
	obj_aa.A_1::n;
	obj_aa.A_2::n;

	return 0;
}
```
