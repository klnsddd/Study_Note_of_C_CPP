假设创建复数类，实例化两个具体的复数对象，直接使用运算符 `+` 并不能实现复数的相加，但可以运算符重载来实现。当然，也可以写个`add`函数来实现
- 运算符重载格式
	`返回值类型 函数名 (形参列表) {}`
	函数名：`operator运算符名称`
	注意一些运算符的左右结合性：`+` `-` 都是左结合性
- 在全局重载运算符
	在类中实现时可以使用this指针
	全局没有this指针则按照正常函数来编写，比上面多写一个参数即可；另外，使用`friend`来访问`private`数据

- 额外知识点
	新建的类中至少有4个默认函数，构造，析构，拷贝构造，以及**赋值函数**

```C++
#include <iostream>
using namespace std;

// 复数：a + bi
class MyComplex
{
public:
	MyComplex(double real = 0.0, double imag = 0.0);
	~MyComplex();
	void display() const;
	
	//方法1
	MyComplex operator+(MyComplex other)
	//隐藏了第一个加数，对应this指针，`obj_1 + `触发该obj_1的该成员函数，`obj_2`传入参数列表
	{
		MyComplex obj;
		obj.m_real = this->m_real + other.m_real;
		obj.m_imag = this->m_imag + other.m_imag;
		return obj;
	}
	
	//方法2
	MyComplex& operator+(const MyComplex& other) const;
	//防止调用拷贝构造函数的方法
	
	//全局
	friend MyComplex& operator+(const MyComplex& obj1, const MyComplex& obj2)；
	
private:
	double m_real;
	double m_image;
}

//在全局实现
MyComplex& operator+(const MyComplex& obj1, const MyComplex& obj2)
{
	return MyComplex(obj1.m_real + obj2.m_real, obj1.m_imag + obj2.m_imag);
}

int main()
{
	/*6 + 1;
	3.14 + 1.73; 
	5 + 1.23;    //这其实就是运算符重载，+号的输入参数是不同类型的
	
	int num = 0;
	char ch = 'A';
	cout << num << endl;
	cout << ch << endl;    //这也是重载*/
	
	MyComplex obj_1(3,5);
	obj_1.display();
	MyComplex obj_2(4,6);
	obj_2.display();
	
	MyComplex obj_3;
	obj_3 = obj_1 + obj_2;
	obj_3.display();
	
	obj_1.operator+(obj_2);  //obj_1 + obj_2
	obj_2.operator+(obj_1);  //obj_2 + obj_1
	
	return 0;
}

MyComplex::MyComplex(double real, double imag)
:m_real(real), m_imag(imag)
{
}

void display() const
{
	cout << "()" << m_real << "+" << m_imag << "i)";
}

MyComplex& MyComplex::operator+(const MyComplex& other) const
{
	return MyComplex(this->m_real + other.m_real, this->m_imag + other.m_imag);
	//利用return的匿名对象不会调用拷贝构造函数的特性
}

MyComplex::~MyComplex()
{
}
```
