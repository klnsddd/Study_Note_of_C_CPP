# 链表

线性表的一种，表中的数据不必按照顺序存储，而是在每一个节点里存储下一个节点指针的数据结构
- 物理位置非连续，逻辑位置连续
- 在实现链式表之前，先确定表中某一节点类型
- 再用结构体指针定义链表
- 双向链表 —— 每一节点既保存后一节点，又保存上一节点
- 循环链表 —— 链表的头尾相连
- 链表的特点
	1. 没有固定的长度，可以自由增加节点
	2. 可以快速插入删除数据
- 链表数组的优劣
	1. 数组查找方便，尾部插入删除数据效率高，头部中部插入删除效率低，需要移位
	2. 链表查找麻烦，在任何位置的插入和删除效率高

#include <stdio.h>
#include <stdlib.h>
struct MyNode
{
	// struct MyNode* pfront;
	int data;
	struct MyNode* pNext;	
}NODE;    //C++结构体重定义方式

//单向链表的类型定义
typedef struct LinkList
{
	NODE* pHead; //链表的头节点
	NODE* pEnd;  //链表的尾节点
	int length;  //链表的长度
}LL;     //C结构体重定义方式

//创建单链表
LL* initList()
{
	LL* pTemp = (LL*)malloc(sizeof(LL));
	if (NULL = pTemp)
	{
		return NULL;
	}
	pTemp->pHead = NULL;
	pTemp->pEnd = NULL;
	pTemp->length = 0;
	return pTemp;
}

//创建链表的一个节点
NODE* initNode(int data)
{
	NODE* pTemp = (NODE*)malloc(sizeof(NODE));
	if (NULL = pTemp)
	{
		return NULL;
	}
	pTemp->data = data;
	pTemp->pNext = NULL;
	return pTemp;
}

//链表的死亡
void delList(LL** pList)
{
	if (NULL == *pList)
	{
		printf("链表不存在\n");
		return;
	}
	NODE* pTemp = NULL;
	while ((*pList)->pHead)
	{
		pTemp = (*pList)->pHead;
		(*pList)->pHead = (*pList)->pHead->pNext; 
		//pList结构体中有pHead成员，而pHead指向的是一个结构体，可再次访问其成员pNext
		//指向下一个节点，并在下一步释放当前节点，不断重复此过程来释放所有节点
		free(pTemp);
	}
	free(*pList);
	*pList = NULL;
}

//尾部添加
void appendList(LL* pList, int val)
{
	if (NULL == pList)
	{
		printf("链表不存在, 无法添加\n");
		return;
	}
	NODE *pTemp = initNode(val);
	if (NULL == pTemp)
	{
		printf("节点分配失败，无法添加\n");
		return;
	}
	
	if (NULL == pList->pHead) //空链表
	{
		pList->pHead = pList->pEnd = pTemp;
		//空链表中追加节点，既是头也是尾
	}
	else
	{
		pList->pEnd->pNext = pTemp;
		pList->pEnd = pTemp;
	}
	pList->length++;
}

void insertList(LL* pList, int idx, int val)
{
	if (NULL == pList)
	{
		printf("链表不存在, 无法添加\n");
		return;
	}
	NODE *pTemp = initNode(val);
	if (NULL == pTemp)
	{
		printf("节点分配失败，无法添加\n");
		return;
	}

	//规定，插入的下标超出元素个数，直接追加在尾部
	if (idx >= pList->length)
	{
		pList->pEnd->pNext = pTemp;
		pList->pEnd = pTemp;
	}
	//规定，插入的下标小于等于0，在头部插入
	else if (idx <= 0)
	{
		pTemp->pNext = pList->pHead;
		pList->pHead = pTemp;
	}
	else //中间插入
	{
		NODE* pInsertNode = pList->pHead;
		for (int i=0; i<idx-1; ++i)
		{
			pInsertNode = pInsertNode->pNext;
		}
		pTemp->pNext = pInsertNode->pNext;
		pInsertNode->pNext = pTemp;
	}
	pList->length++;
}

//删除节点
void deleteNode(LL* pList, int idx)
{
	if (NULL == pList || 0 == pList->length)
	{
		printf("链表不存在或为空, 无法删除\n");
		return;
	}
	if (idx < 0 || idx >= pList->length)
	{
		printf("删除位置错误\n");
		return;
	}
	NODE* pTemp = NULL;
	if (0 == idx)
	{
		pTemp = pList->pHead;
		pList->pHead = pTemp->pNext;
	}
	else
	{
		NODE *pCurNode = pList->pHead;
		for (int i=0; i<idx-1; ++i)
		{
			pCurNode = pCurNode->pNext;
		}
		pTemp = pCurNode->pNext;
		pCurNode->pNext = pTemp->pNext;
		if (NULL == pCurNode->pNext)
			pList->pEnd = pCurNode;
	}
	free(pTemp);
	pList->length--;
}
