
- 动态内存分配
	程序员根据需求，向系统主动申请所需大小的内存空间
- 头文件  `stdlib.h`
- 操作相关函数
	malloc    分配一个大小为size的内存区域
	calloc    分配n个大小为size的连续内存区域
	realloc    重新分配通过malloc或calloc开辟的内存空间，改变其内存空间大小
	free    用于释放上述三个函数分配的动态内存空间，**一定不能忘！！！**
- 分配的内存在堆区
	堆区特性：由程序员主动分配，也应该由程序员主动释放
- 数据在内存中无类型，使用内存数据时，通过类型去内存中取相应的内存大小的值，转成类型的数据

**注意：**
- C语言中不需要强制类型转换
	`Arr* pArr = malloc(sizeof(Arr));`
- C++中需要强制类型转换
	`Arr* pArr = (Arr*)malloc(sizeof(Arr));`

```C++
#include <stdlib.h>
int main(void)
{
	int* p = NULL, * q = NULL;
	p = (int*)malloc(sizeof(int)); //分配一个4字节大小的内存区域，把首地址当成int*地址赋值给到p 
	*p = 123;
	printf("%d\n",*p);
	free(p);
	
	q = (int*)calloc(4, sizeof(int)); //分配一个16字节大小的内存区域，把首地址当成int*地址赋值给到q, q相当于指向了了一个没有名字的数组
	q[0]=1;
	q[1]=2;
	for (int i=0; i<4; i++)
		printf("%d\n",*(q+i));
	
	realloc(q, sizeof(int)*10);
	q[6] = 123;
	q[9] = 99;
	free(q);
	
	int a = 4407873; //0x00434241
	printf("%s\n",(char*)&a); //%s需要的是字符串的首地址，所以先取地址用&，然后再转为字符类型以匹配%s，所以要强制转换类型，又由于需要的是地址，所以是(char*)
	//或者这样解释 &a的类型时int*，是指针，我们要将int指针转为char指针，即将原本为int类型的地址看成是char类型的地址
	// 输出：ABC
	char *p = (char*)malloc(4);
    p[0] = 'a';
    p[1] = 'b';
    p[2] = 'c';
    p[3] = '\0';
    printf("%s\n", p);
    free(p); 
	return 0;
}
```

```C++
#include <stdio.h>
#include <stdlib.h>

int* getMem(int num)  //此处，指针作为函数的返回值就有意义了
{
	int* p = (int*)malloc(num);
	return p;
}

int main(void)
{
    int *p = getMem(10);
    for(int i=0; i<10; i++)
        p[i] = i+1;
    for(int i=0; i<10; i++)
        printf("%d ", *(p+i));
    free(p);

    return 0;

}
```
