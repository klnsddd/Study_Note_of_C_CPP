排序规则：在序列中找到最大或最小值，将其移动到首位或末位。
n个数字
做n-1轮比较
第k次比较n-k-1次(从0开始计次)

```
3 1 5 2 4
1 3 5 2 4
1 3 5
1 2 5 3 4
1 2 3 5 4
1 2 3 4 5
```
# 方法一
每次遇到最大值或最小值都移动
```C++
int main(void)
{
	int arr[5] = {6,9,3,1,8};
	int n=5;
	for (int i=0; i<n-1; i++)
	{
		for (int j=i+1; j<n-1; j++)
		{
			if (arr[i] > arr[j])
			{
				int tempVal = arr[i];
				arr[i] = arr[j];
				arr[j] = tempVal;
			}
		}
	}
	for (int i=0; i<n; i++)
		printf("%d\t", arr[i]);
	return 0;
}

```
# 方法二
先找到最大值或最小值，然后每轮只做一次交换
```C++
int main(void)
{
	int arr[5] = {6,9,3,1,8};
	int n=5;
	for (int i=0; i<n-1; i++)
	{
		int minIdx = i;
		for (int j=i+1; j<n; j++)
		{
			if (arr[j]<arr[minIdx])
				minIdx = j;
		}
		if (minIdx != i)
			tempVal = arr[i];
			arr[i] = arr[minIdx];
			arr[minIdx] = tempVal;
	}
	return 0;
}
```
