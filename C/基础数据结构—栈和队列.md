栈和队列
	操作受限的线性表
栈
	一端开口的玻璃杯，先进后出
	可操作的一端叫栈顶，不可操作的一段叫栈底
	操作：入栈，出栈，得到栈顶元素，判断栈满或栈空
	顺序表更方便表示栈
队列
	食堂窗口排队，先进先出
	操作：入队，出队，得到队头，得到队尾，判断队空（队满没必要）
	链表更适合

# 栈
```C++
#include <stdio.h>
#include <stdlib.h>
#define STACK_LEN 10
typedef struct MyStack  //顺序结构的栈
{
	int dataArr[STACK_LEN];
	int top;
}STACK;

//创建
STACK* initStack()
{
	STACK* pStack = (STACK*)malloc(sizeof(STACK));
	if (NULL == pStack)
		return NULL;
	pStack->top = 0;
	return pStack;
}

//判断栈满
int fullStack(STACK* pStack)
{
	if (NULL == pStack)
	{
		printf("栈不存在\n");
		return 0;
	}
	if (pStack->top >= STACK_LEN)
		return 1;
	return 0;
}

//判断栈空
int emptyStack(STACK* pStack)
{if (NULL == pStack)
	{
		printf("栈不存在\n");
		return 0;
	}
	if (0 == pStack->top)
		return 1;
	return 0;
}

//入栈
void pushStack(STACK* pStack, int val)
{
	if (NULL == pStack)
	{
		printf("栈不存在，不能入栈\n");
		return;
	} //此处注意与判断栈满栈空函数有冗余部分
	if (1 == fullStack(pStack))
	{
		printf("栈满，不能入栈\n");
		return;
	}
	pStack->dataArr[pStack->top++] = val;
}

//出栈
void popStack(STACK* pStack)
{
	if (NULL == pStack)
	{
		printf("栈不存在，不能出栈\n");
		return;
	} //此处注意与判断栈满栈空函数有冗余部分
	if (1 == emptyStack(pStack))
	{
		printf("栈空，不能出栈\n");
		return;
	}
	pStack->top--;
}

//获取栈顶元素
int topStack(STACK* pStack)
{
	//无法写函数判断栈空或栈不存在，返回值有可能和栈中的值相等，若栈空或栈不存在，函数报错即可
	return pStack->dataArr[pStack->top - 1];
}

void clearStack(STACK** pStack)
{
	if (NULL != *pStack)
	{
		free(*pStack);
		*pStack = NULL;
	}
}

int main(void)
{
	STACK* pStack = NULL;
	pStack = initStack();
	pushStack(pStack, 1);
	pushStack(pStack, 2);
	popStack(pStack);
	pushStack(pStack, 3);
	int val = topStack(pStack);
	
	clearStack(&pStack);
	return 0;
}
```
