# 线性表

表中数据元素之间的关系是一对一的关系，数据成线性的方式来分布，比如说排队

# 顺序表

- 线性表的一种，是将表中的数据依次存放在计算机内存中一组地址连续的存储单元中的一种数据结构
- 物理位置连续（例如数组位置1、2、3 ...），逻辑位置连续（例如数据依次存放到数组位置1、2、3 ...）
- 基本操作
	增、删、改、查

## 数组（定长的顺序表）

数组是顺序表的一种表现，其长度在定义时就固定了
- 利用表示数组长度的变量如len来便于操作
- 数组做增加时
	- 首先要有足够的空间，然后可以在尾部，中部，或头部增加
	- 若在中部或头部增加，需要把插入位置后面的数据后移，从末尾开始后移
	- 若空间不够要做插入，会导致后面的数据的溢出
- 数组做删除时
	- 在尾部删除时，直接控制表示数据元素长度的变量使其不能访问
	- 中部或头部删除时，后面数据前移覆盖要删除的数据，长度--
```C++
int main(void)
{
	int arr[6] = {10,20,30,40,50};
	int len = 5;
	// arr[5] = 60; //尾部增加
	//在中部或头部增加时，需要把插入位置后面的数据后移，从末尾开始后移
	int idx = 3;
	for(int i=len-1; i>=idx; --i)
	{
		arr[i+1] = arr[i];
	}
	arr[idx] = 60;
	len++;
	
	//尾部删除
	//len--;
	//中部或头部删除
	int delIdx = 3;
	for (int i=delIdx; i<len-1; ++i)
	{
		arr[i] = arr[i+1];
	}
	len--;
	
	//查找
	int findVal = 20;
	char isFind = 'N';
	for (int i=0; i<len; i++)
	{
		if (findVal == arr[i])
			{
				printf("Found it, subscript is %d\n", i);
				isFind = 'Y';
				break;
			}
	}
	if (isFind == 'N')
		printf("Didn't find it\n");
	
	return 0;
}
```

## 顺序表（不定长、动态数组）

```C++
#include <stdio.h>
#include <stdlib.h>
typedef struct MyArray
{
	int* pData; //存储数据的顺序表
	int length;
}Arr;

//创建顺序表
Arr* createArr(void)
{
	Arr* pArr = (Arr*)malloc(sizeof(Arr));
	if (NULL == pArr)
	{
		printf("创建顺序表失败\n");
		return NULL;
	}
	pArr->pData = NULL;
	pArr->length = 0;
	return pArr;
}

// 释放顺序表
void freeArr(Arr** pArr)
{
    if (NULL == *pArr)
    {
        printf("顺序表指针为空\n");
        return;
    }

    if (NULL != (*pArr)->pData)
        free((*pArr)->pData); //pData之后也会被分配内存，需要先清除，否则清除顺序表后将无法访问pData的内存，也无法清除

    free(*pArr);
    *pArr = NULL;
}

//清空顺序表
void clearArr(Arr* pArr)
{
	if(NULL == pArr)
	{
		printf("顺序表为空，不能清除\n");
		return;
	}
	if(NULL == pArr->pData)
	{
		printf("顺序表数据为空，不能清除\n");
		return;
	}
	free(pArr->pData);
	pArr->pData = NULL;
	pArr->length = 0;
}

//判断顺序表是否为空
int emptyArr(Arr* pArr)
{
	if(NULL == pArr)
	{
		printf("顺序表不存在\n");
		return 1;
	}
	if(NULL == pArr->length)
	{
		printf("顺序表为空\n");
		return 1;
	}
	return 0;
}

//添加数据(尾部追加)
void pushArr(Arr* pArr, int elem)
{
	if(NULL == pArr)
	{
		printf("顺序表不存在, 无法追加数据\n");
		return;
	}
	pArr->length++;
	int* pTempArr = (int*)malloc(sizeof(int)*(pArr->length));
	for (int i=0; i<(pArr->length)-1; ++i)
	{
		pTempArr[i] = pArr->pData[i];
	}
	if (NULL != pArr->pData)
	{
		free(pArr->pData);
	}
	pArr->pData = pTempArr;
	pArr->pData[pArr->length-1] = elem;
}

//添加数据(中前部追加)
void insertArr(Arr* pArr, int idx, int elem)
{
	if(NULL == pArr)
	{
		printf("顺序表不存在, 无法追加数据\n");
		return;
	}
	if (idx<0 || idx>=pArr->length)
	{
		printf("插入位置错误，无法插入数据\n");
		return;
	}
	pArr->length++;
	int* pTempArr = (int*)malloc(sizeof(int)*(pArr->length));
	for (int i=0; i<(pArr->length)-1; ++i)
	{
		pTempArr[i] = pArr->pData[i];
	}
	if (NULL != pArr->pData)
	{
		free(pArr->pData);
	}
	pArr->pData = pTempArr;
	for(int i=pArr->length-1; i>idx; --i)
	{
		pArr->pData[i] = pArr->pData[i-1];
	}
	pArr->pData[idx] = elem;
}

//删除数据(尾部删除)
void popArr(Arr* pArr)
{
	if(NULL == pArr)
	{
		printf("顺序表不存在, 无法删除数据\n");
		return;
	}
	if(NULL == pArr->pData)
	{
		printf("顺序表数据为空，无法删除数据\n");
		return;
	}
	pArr->length--;
}

//删除数据(前中部删除)
void delArr(Arr* pArr, int idx)
{
	if(NULL == pArr)
	{
		printf("顺序表不存在, 无法删除数据\n");
		return;
	}
	if(NULL == pArr->pData)
	{
		printf("顺序表数据为空，无法删除数据\n");
		return;
	}
	if (idx<0 || idx>=pArr->length)
	{
		printf("删除位置错误，无法删除数据\n");
		return;
	}
	for (int i=idx; i<pArr->length-1; i++)
	{
		pArr->pData[i] = pArr->pData[i+1];
	}
	pArr->length--;
}

//查找数据
int findArr(Arr* pArr, int findVal)
{
	if(NULL == pArr)
	{
		printf("顺序表不存在, 无法查找数据\n");
		return -1;
	}
	if(NULL == pArr->pData)
	{
		printf("顺序表数据为空，无法查找数据\n");
		return -1;
	}
	for (int i=0; i<pArr->length; i++)
	{
		if(pArr->pData[i] == findVal)
			return i;
	}
	return -1;
}

//改写数据
void changeArr(Arr* pArr, int idx, int changVal)
{
	if(NULL == pArr)
	{
		printf("顺序表不存在, 无法改写数据\n");
		return;
	}
	if(NULL == pArr->pData)
	{
		printf("顺序表数据为空，无法改写数据\n");
		return;
	}
	if (idx<0 || idx>=pArr->length)
	{
		printf("改写位置错误，无法改写数据\n");
		return;
	}
	pArr->pData[idx] = changVal;
}

//测试遍历顺序表
void printfArr(Arr* pArr)
{
	if(NULL == pArr)
	{
		printf("顺序表不存在, 无法遍历数据\n");
		return;
	}
	if(NULL == pArr->pData)
	{
		printf("顺序表数据为空，无法遍历数据\n");
		return;
	}
	for (int i=0; i<pArr->length; ++i)
	{
		printf("%d\t", pArr->pData[i]);
	}
	printf("\n");
}

int main(void)
{
	Arr* pData = createArr();
	pushArr(pData, 1);
	pushArr(pData, 2);
	insertArr(pData, 1, 3);
	//popArr(pData);
	//delArr(pData, 1);
	int idx = findArr(pData, 3);
	changArr(pData, idx ,999);
	printfArr(pData);
	freeArr(&pData);
	return 0;
}
```
